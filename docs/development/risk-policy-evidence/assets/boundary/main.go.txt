package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"strings"
	"time"

	"github.com/open-policy-agent/opa/v1/ast"
	"github.com/open-policy-agent/opa/v1/rego"
	"github.com/open-policy-agent/opa/v1/topdown"
)

func capabilities() *ast.Capabilities {
	c := ast.CapabilitiesForThisVersion()
	names := strings.Fields("eq equal neq lt lte gt gte count is_array is_object is_string is_number is_boolean is_null object.get object.keys array.concat concat sprintf startswith endswith contains lower upper to_number plus minus mul div")
	all := c.Builtins
	c.Builtins = nil
	for _, name := range names {
		found := false
		for _, b := range all {
			if b.Name == name {
				c.Builtins = append(c.Builtins, b)
				found = true
				break
			}
		}
		if !found {
			panic("missing builtin " + name)
		}
	}
	c.AllowNet = []string{}
	return c
}

// Only fixture shapes with empty paths are accepted; this is not the product validator.
func validate(v any) string {
	m, ok := v.(map[string]any)
	if !ok || len(m) != 2 {
		return "invalid"
	}
	level, ok := m["level"].(string)
	if !ok || !strings.Contains("|none|low|medium|high|critical|", "|"+level+"|") {
		return "invalid"
	}
	rs, ok := m["reasons"].([]any)
	if !ok || len(rs) < 1 || len(rs) > 16 {
		return "invalid"
	}
	for _, r := range rs {
		rm, ok := r.(map[string]any)
		if !ok || len(rm) != 3 {
			return "invalid"
		}
		code, cok := rm["code"].(string)
		msg, mok := rm["message"].(string)
		paths, pok := rm["paths"].([]any)
		if !cok || code != "synthetic" || !mok || strings.TrimSpace(msg) == "" || !pok || len(paths) != 0 {
			return "invalid"
		}
	}
	return "valid"
}

func main() {
	c := capabilities()
	if os.Args[1] == "capabilities" {
		json.NewEncoder(os.Stdout).Encode(c)
		return
	}
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	var spec struct {
		Source string
		Input  any
		Strict bool
		Reject bool
	}
	if err = json.Unmarshal(data, &spec); err != nil {
		panic(err)
	}
	out := map[string]any{"prepare_error": "", "eval_error": "", "eval_error_code": "", "result_count": 0, "expression_count": 0, "value": nil, "validation": "not_run", "context_cause": "", "print": "", "trace": "", "hooks": "not installed"}
	started := time.Now()
	defer func() {
		out["duration_ns"] = time.Since(started).Nanoseconds()
		b, e := json.MarshalIndent(out, "", "  ")
		if e != nil {
			panic(e)
		}
		if len(b) > 1048576 {
			panic("result overflow")
		}
		if e = os.WriteFile(os.Args[2], b, 0600); e != nil {
			panic(e)
		}
	}()
	module, err := ast.ParseModule("fixture.rego", spec.Source)
	if err != nil {
		out["prepare_error"] = err.Error()
		return
	}
	if spec.Reject {
		denied := false
		ast.WalkExprs(module, func(e *ast.Expr) bool {
			if e.IsCall() {
				s := e.Operator().String()
				if s == "print" || s == "trace" {
					denied = true
				}
			}
			return false
		})
		ast.WalkTerms(module, func(t *ast.Term) bool {
			if call, ok := t.Value.(ast.Call); ok && len(call) > 0 {
				s := call[0].String()
				if s == "print" || s == "trace" {
					denied = true
				}
			}
			return false
		})
		if denied {
			out["prepare_error"] = "source rejected: print/trace"
			out["phase"] = "source_check"
			return
		}
	}
	ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
	defer cancel()
	q, err := rego.New(rego.Query("data.tfui.risk.v1.classification"), rego.ParsedModule(module), rego.Capabilities(c), rego.StrictBuiltinErrors(spec.Strict), rego.EnablePrintStatements(false)).PrepareForEval(ctx)
	if err != nil {
		out["prepare_error"] = err.Error()
		out["context_cause"] = fmt.Sprint(context.Cause(ctx))
		return
	}
	rs, err := q.Eval(ctx, rego.EvalInput(spec.Input))
	out["context_cause"] = fmt.Sprint(context.Cause(ctx))
	if err != nil {
		out["eval_error"] = err.Error()
		if e, ok := err.(*topdown.Error); ok {
			out["eval_error_code"] = e.Code
		}
		return
	}
	out["result_count"] = len(rs)
	if len(rs) == 0 {
		out["validation"] = "undefined"
		return
	}
	out["expression_count"] = len(rs[0].Expressions)
	if len(rs) != 1 || len(rs[0].Expressions) != 1 {
		out["validation"] = "cardinality"
		return
	}
	v := rs[0].Expressions[0].Value
	out["value"] = v
	out["value_type"] = fmt.Sprintf("%T", v)
	out["validation"] = validate(v)
}
