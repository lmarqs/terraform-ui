package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"time"

	"github.com/open-policy-agent/opa/v1/ast"
	"github.com/open-policy-agent/opa/v1/rego"
)

func main() { d03Main(os.Args[1:]) }

func d03Main(args []string) {
	if len(args) < 5 || len(args) > 6 {
		panic("usage: fixtures capabilities N cold|warm candidate|standalone|overhead [cancelled|deadline]")
	}
	n, err := strconv.Atoi(args[2])
	if err != nil || (n != 1 && n != 100 && n != 1000) {
		panic("invalid bounded workload")
	}
	phase, variant := args[3], args[4]
	if phase != "cold" && phase != "warm" {
		panic("invalid phase")
	}
	if variant != "candidate" && variant != "standalone" && variant != "overhead" {
		panic("invalid variant")
	}
	mode := ""
	if len(args) == 6 {
		mode = args[5]
		if mode != "cancelled" && mode != "deadline" {
			panic("invalid cancellation mode")
		}
	}
	out := map[string]any{"workload_n": n, "phase": phase, "variant": variant, "prepare_count": 0, "parse_ns": int64(0), "source_parse_ns": int64(0), "prepare_ns": int64(0), "first_eval_ns": int64(0), "eval_total_ns": int64(0), "correct_count": 0, "eval_count": 0, "loops": 1}
	started := time.Now()
	fixtureBytes, err := os.ReadFile(args[0])
	d03Require(err)
	var fixture struct {
		Minimal       map[string]any `json:"minimal"`
		Policy        string         `json:"policy"`
		BoundedPolicy string         `json:"bounded_policy"`
	}
	decoder := json.NewDecoder(bytes.NewReader(fixtureBytes))
	decoder.UseNumber()
	d03Require(decoder.Decode(&fixture))
	inputs := make([]any, n)
	for i := range inputs {
		b, err := json.Marshal(fixture.Minimal)
		d03Require(err)
		var input map[string]any
		d := json.NewDecoder(bytes.NewReader(b))
		d.UseNumber()
		d03Require(d.Decode(&input))
		resource, ok := input["resource"].(map[string]any)
		if !ok {
			panic("fixture has no resource")
		}
		resource["address"] = fmt.Sprintf("example_item.synthetic_%d", i)
		inputs[i] = input
	}
	serialized, err := json.Marshal(inputs)
	d03Require(err)
	if len(serialized) > 1<<20 {
		panic("input exceeds 1MiB")
	}
	out["input_bytes"] = len(serialized)
	out["parse_ns"] = time.Since(started).Nanoseconds()
	var caps ast.Capabilities
	capsBytes, err := os.ReadFile(args[1])
	d03Require(err)
	d03Require(json.Unmarshal(capsBytes, &caps))
	policy := fixture.Policy
	if mode != "" {
		policy = fixture.BoundedPolicy
	}
	if len(policy) > 64<<10 {
		panic("policy exceeds 64KiB")
	}
	var query rego.PreparedEvalQuery
	if variant != "overhead" {
		start := time.Now()
		module, err := ast.ParseModule("cost.rego", policy)
		d03Require(err)
		out["source_parse_ns"] = time.Since(start).Nanoseconds()
		prepareCtx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
		start = time.Now()
		query, err = rego.New(rego.Query("data.tfui.risk.v1.classification"), rego.ParsedModule(module), rego.Capabilities(&caps), rego.StrictBuiltinErrors(true), rego.EnablePrintStatements(false)).PrepareForEval(prepareCtx)
		cancel()
		d03Require(err)
		out["prepare_count"] = 1
		out["prepare_ns"] = time.Since(start).Nanoseconds()
	}
	expected := map[string]any{"level": "low", "reasons": []any{map[string]any{"code": "synthetic", "message": "synthetic update", "paths": []any{}}}}
	evaluate := func(input any) (int64, error) {
		if variant == "overhead" {
			start := time.Now()
			m, ok := input.(map[string]any)
			if !ok || m["resource"] == nil || m["change"] == nil {
				panic("invalid dispatch input")
			}
			return time.Since(start).Nanoseconds(), nil
		}
		ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
		if mode == "cancelled" {
			cancel()
		}
		defer cancel()
		option := rego.EvalInput(input)
		start := time.Now()
		result, err := query.Eval(ctx, option)
		elapsed := time.Since(start).Nanoseconds()
		if mode != "" {
			out["context_cause"] = fmt.Sprint(context.Cause(ctx))
			out["result_count"] = len(result)
		}
		if err != nil {
			return elapsed, err
		}
		if len(result) != 1 || len(result[0].Expressions) != 1 || !reflect.DeepEqual(result[0].Expressions[0].Value, expected) {
			panic("incorrect classification or result cardinality")
		}
		return elapsed, nil
	}
	if mode != "" {
		elapsed, err := evaluate(inputs[0])
		out["eval_total_ns"] = elapsed
		out["mode"] = mode
		out["eval_error"] = fmt.Sprint(err)
		out["outcome"] = "success"
		if err != nil {
			out["outcome"] = "eval_error"
		}
		if mode == "cancelled" && err == nil {
			panic("cancelled evaluation succeeded")
		}
		d03Require(json.NewEncoder(os.Stdout).Encode(out))
		return
	}
	loops := 1
	if phase == "warm" {
		loops = 100
		_, err := evaluate(inputs[0])
		d03Require(err)
	}
	out["loops"] = loops
	correct := 0
	var total int64
	for pass := 0; pass < loops; pass++ {
		for _, input := range inputs {
			elapsed, err := evaluate(input)
			d03Require(err)
			total += elapsed
			if correct == 0 {
				out["first_eval_ns"] = elapsed
			}
			correct++
		}
	}
	out["eval_total_ns"] = total
	out["correct_count"], out["eval_count"] = correct, correct
	d03Require(json.NewEncoder(os.Stdout).Encode(out))
}

func d03Require(err error) {
	if err != nil {
		panic(err)
	}
}
