func TestContextSource_GivenRealHCL_ShouldExposeCurrentSelectorLimitations(t *testing.T) {
	root := t.TempDir()
	writeTfuiHCL(t, root, `defaults {
 plugin "risk" {
   source = "policies/root.rego"
   empty = ""
   invalid = missing.value
 }
 risk_policy = "policies/dedicated.rego"
}`)
	cfg, err := config.LoadRoot(root)
	if err != nil {
		t.Fatal(err)
	}
	opts := config.Resolve(cfg, nil, "prod").PluginConfig("risk").Options
	if opts["source"] != "policies/root.rego" || opts["empty"] != "" {
		t.Fatalf("options lost literal values: %#v", opts)
	}
	if _, ok := opts["invalid"]; ok {
		t.Fatalf("invalid expression unexpectedly retained: %#v", opts)
	}
	t.Logf("root generic option path remains unresolved literal; empty retained, invalid expression discarded: %#v", opts)
	for _, body := range []string{`plugin "risk" { source = "x" }`, `workspace "prod" {
 plugin "risk" { source = "x" }
}`} {
		member := filepath.Join(root, "member")
		writeTfuiHCL(t, member, body)
		_, err := config.LoadChild(member)
		if err == nil || !strings.Contains(err.Error(), "root tfui.hcl") {
			t.Fatalf("Member plugin block not explicitly rejected: %v", err)
		}
		t.Logf("Member plugin rejection: %v", err)
	}
	for _, body := range []string{`risk_policy = "x"`, `risk_policy = ""`, `risk_policy = null`, `risk_policy = 1`} {
		member := filepath.Join(root, "member")
		writeTfuiHCL(t, member, body)
		if _, err := config.LoadChild(member); err == nil {
			t.Fatalf("dedicated selector unexpectedly supported: %s", body)
		}
	}
	writeTfuiHCL(t, root, `defaults {
 plugin "risk" {}
}`)
	cfg, err = config.LoadRoot(root)
	if err != nil {
		t.Fatal(err)
	}
	if len(config.Resolve(cfg, nil, "").PluginConfig("risk").Options) != 0 {
		t.Fatal("absent selector must remain absent")
	}
	writeTfuiHCL(t, root, `defaults {`)
	if _, err = config.LoadRoot(root); err == nil {
		t.Fatal("malformed root config accepted")
	}
}

func TestContextSource_GivenStartupAndSwitch_ShouldTraceRealFallbackAndWorkspacePrecedence(t *testing.T) {
	root := t.TempDir()
	writeTfuiHCL(t, root, `member "A" {}
member "B" {}
defaults {
 var_file "root.tfvars" {}
}`)
	writeTfuiHCL(t, filepath.Join(root, "A"), `var_file "a.tfvars" {}
workspace "p*" { lock_timeout = "1m" }
workspace "prod*" { lock_timeout = "2m" }
workspace "prod" { lock_timeout = "3m" }`)
	writeTfuiHCL(t, filepath.Join(root, "B"), `plugin "risk" { source = "bad" }`)
	rootCfg, err := config.LoadRoot(root)
	if err != nil {
		t.Fatal(err)
	}
	registry := plugin.NewRegistry()
	registry.Build(nil, nil)
	cfg := config.Config{Dir: root, Chdir: "A"}
	app := NewApp(cfg, newMockService("prod", nil), registry, rootCfg)
	ctx := app.startupContext("prod")
	if ctx.Chdir != "A" || ctx.WorkingDir != filepath.Join(root, "A") || ctx.LockTimeout != "3m" || !reflect.DeepEqual(ctx.VarFiles, []string{"root.tfvars", "a.tfvars"}) {
		t.Fatalf("startup wrong Context: %#v", ctx)
	}
	t.Logf("startup selected A/prod: chdir=%q vars=%v timeout=%q", ctx.Chdir, ctx.VarFiles, ctx.LockTimeout)
	for _, tt := range []struct {
		workspace sdk.Workspace
		want      sdk.LockTimeout
	}{{"preview", "1m"}, {"prod", "3m"}} {
		model, _ := app.Update(sdk.ContextSwitchRequestMsg{Chdir: "A", Workspace: tt.workspace})
		app = model.(App)
		if app.holder.current.LockTimeout != tt.want {
			t.Fatalf("workspace %q timeout=%q want=%q", tt.workspace, app.holder.current.LockTimeout, tt.want)
		}
	}
	model, _ := app.Update(sdk.ContextSwitchRequestMsg{Chdir: "B", Workspace: "prod"})
	app = model.(App)
	if app.childCfg != nil || !reflect.DeepEqual(app.holder.current.VarFiles, []string{"root.tfvars"}) {
		t.Fatalf("expected observed invalid-child fallback: %#v", app.holder.current)
	}
	t.Log("switch A -> invalid B falls back to root execution settings; policy integration must not inherit this failure behavior")
	cfg.Chdir = "B"
	startup := NewApp(cfg, newMockService("prod", nil), registry, rootCfg)
	if startup.childCfg != nil || !reflect.DeepEqual(startup.startupContext("prod").VarFiles, []string{"root.tfvars"}) {
		t.Fatal("invalid startup child did not exhibit root fallback")
	}
	// A root-as-Member is currently parsed as child too, then rejected due to defaults.
	cfg.Chdir = ""
	rootApp := NewApp(cfg, newMockService("prod", nil), registry, rootCfg)
	if rootApp.childCfg != nil {
		t.Fatal("expected root-as-child parse rejection")
	}
	noConfig := t.TempDir()
	cfg = config.Config{Dir: noConfig, Chdir: "A"}
	if err := os.MkdirAll(filepath.Join(noConfig, "A"), 0755); err != nil {
		t.Fatal(err)
	}
	zero := NewApp(cfg, newMockService("prod", nil), registry, nil)
	zctx := zero.startupContext("prod")
	if zctx.Chdir != "A" || zctx.WorkingDir != filepath.Join(noConfig, "A") || len(zctx.VarFiles) != 0 {
		t.Fatalf("zero-config startup wrong: %#v", zctx)
	}
	t.Log("zero-config startup retains selected Chdir and no policy selection; no parent discovery proved here")
}
